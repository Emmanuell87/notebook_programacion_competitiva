\textbf{Multiplicación segura con mod:}
\[
(a \cdot b) \bmod m = \left( (a \bmod m) \cdot (b \bmod m) \right) \bmod m
\]

\textbf{Propiedades útiles:}
\begin{itemize}
  \item \( (a + b) \bmod m = ((a \bmod m) + (b \bmod m)) \bmod m \)
  \item \( (a - b) \bmod m = ((a \bmod m) - (b \bmod m) + m) \bmod m \)
  \item \( a^b \bmod m \) se puede calcular con \textbf{potenciación binaria}.
\end{itemize}
